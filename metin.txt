local function ShowNotification(message)
    local NotifFrame = Instance.new("Frame")
    NotifFrame.Size = UDim2.new(0, 180, 0, 40)
    NotifFrame.Position = UDim2.new(1, 10, 0.5, -20) -- Sağ tarafta gizli başlar
    NotifFrame.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    NotifFrame.BorderSizePixel = 0
    NotifFrame.Parent = MainFrame

    local NotifCorner = Instance.new("UICorner")
    NotifCorner.CornerRadius = UDim.new(0, 6)
    NotifCorner.Parent = NotifFrame

    local NotifText = Instance.new("TextLabel")
    NotifText.Size = UDim2.new(1, 0, 1, 0)
    NotifText.BackgroundTransparency = 1
    NotifText.Text = message
    NotifText.TextColor3 = Color3.fromRGB(0, 255, 128)
    NotifText.TextSize = 12
    NotifText.Font = Enum.Font.GothamBold
    NotifText.Parent = NotifFrame

    -- Kayma Animasyonu (İçeri Kayma)
    TweenService:Create(NotifFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Position = UDim2.new(1, 15, 0.5, -20)
    }):Play()

    -- 2 saniye sonra kaybolma
    task.delay(2, function()
        local tweenOut = TweenService:Create(NotifFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
            Position = UDim2.new(1, 10, 0.5, -20)
        })
        tweenOut:Play()
        tweenOut.Completed:Connect(function()
            NotifFrame:Destroy()
        end)
    end)
end

-- ESP Yönetimi
local espConnections = {}
local isEnabled = true
local isRunning = true
local healthConnections = {}

local function removeESP(player)
    if player.Character and player.Character:FindFirstChild("TeamESP") then
        player.Character.TeamESP:Destroy()
    end
    if healthConnections[player] then
        healthConnections[player]:Disconnect()
        healthConnections[player] = nil
    end
end

local function createESP(player)
    if player == LocalPlayer then return end
    
    local function addVisuals(character)
        if not isRunning then return end
        if not character:WaitForChild("HumanoidRootPart", 5) then return end
        
        if character:FindFirstChild("TeamESP") then
            character.TeamESP:Destroy()
        end
        
        local folder = Instance.new("Folder")
        folder.Name = "TeamESP"
        folder.Parent = character
        
        -- Kutu (Highlight)
        local highlight = Instance.new("Highlight")
        highlight.Name = "BoxESP"
        highlight.Adornee = character
        highlight.FillTransparency = 0.5
        highlight.OutlineTransparency = 0
        highlight.Enabled = isEnabled
        highlight.Parent = folder
        
        -- Çok Küçük İsim ve Can Etiketi (BillboardGui)
        local billboard = Instance.new("BillboardGui")
        billboard.Name = "NameESP"
        billboard.Adornee = character:WaitForChild("Head")
        billboard.Size = UDim2.new(0, 80, 0, 30)
        billboard.StudsOffset = Vector3.new(0, 1.8, 0)
        billboard.AlwaysOnTop = true
        billboard.Enabled = isEnabled
        billboard.Parent = folder
        
        local textLabel = Instance.new("TextLabel")
        textLabel.Size = UDim2.new(1, 0, 1, 0)
        textLabel.BackgroundTransparency = 1
        textLabel.TextSize = 9 -- Çok küçük yazı boyutu
        textLabel.Font = Enum.Font.SourceSansBold
        textLabel.TextWrapped = true
        textLabel.Parent = billboard
        
        -- Can Güncelleme Mantığı
        local humanoid = character:WaitForChild("Humanoid", 5)
        
        local function updateVisuals()
            if not isRunning then return end
            
            -- Takım Renkleri
            local fillColor = Color3.fromRGB(255, 255, 0)
            local textColor = Color3.fromRGB(255, 255, 0)
            
            if player.Team and LocalPlayer.Team then
                if player.Team == LocalPlayer.Team then
                    fillColor = Color3.fromRGB(0, 255, 0) -- Takım arkadaşı yeşil
                    textColor = Color3.fromRGB(0, 255, 0)
                else
                    fillColor = Color3.fromRGB(255, 0, 0) -- Düşman kırmızı
                    textColor = Color3.fromRGB(255, 0, 0)
                end
            end
            
            highlight.FillColor = fillColor
            
            -- Yazı İçeriği (İsim + Can)
            if humanoid then
                local hp = math.floor(humanoid.Health)
                local maxHp = math.floor(humanoid.MaxHealth)
                textLabel.Text = player.Name .. "\n[" .. hp .. "/" .. maxHp .. "]"
            else
                textLabel.Text = player.Name
            end
            
            textLabel.TextColor3 = textColor
        end
        
        updateVisuals()
        
        if humanoid then
            healthConnections[player] = humanoid.HealthChanged:Connect(updateVisuals)
        end
        
        player:GetPropertyChangedSignal("Team"):Connect(updateVisuals)
        LocalPlayer:GetPropertyChangedSignal("Team"):Connect(updateVisuals)
    end
    
    local conn = player.CharacterAdded:Connect(addVisuals)
    espConnections[player] = conn
    
    if player.Character then
        task.spawn(function()
            addVisuals(player.Character)
        end)
    end
end

-- Oyuncuları Tara
for _, player in ipairs(Players:GetPlayers()) do
    createESP(player)
end

    if espConnections[player] then
        espConnections[player]:Disconnect()

    end

end)

-- Key Verification Handling
SubmitBtn.MouseButton1Click:Connect(function()
    local userKey = TextBox.Text
    
    local success, result = pcall(function()
        return game:HttpGet(KeyListURL)
    end)
    
    if success then
        local isValid = false
        for key in string.gmatch(result, "[^\r\n]+") do
            if string.match(key, "^%s*(.-)%s*$") == userKey then
                isValid = true
                break
            end
        end
        
        if isValid then
            SubmitBtn.Text = "Success!"
            SubmitBtn.BackgroundColor3 = Color3.fromRGB(0, 255, 128)
            task.wait(1)
            StartMainScript()
        else
            SubmitBtn.Text = "Invalid Key!"
            SubmitBtn.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
            task.wait(1.5)
            SubmitBtn.Text = "Submit Key"
            SubmitBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
        end
    else
        SubmitBtn.Text = "Connection Error!"
        task.wait(1.5)
        SubmitBtn.Text = "Submit Key"
    end
end)
